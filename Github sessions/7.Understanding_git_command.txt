git checkout -f  --> this will all rrestore your file to recent commit

git checkout file_path --> this will restore a specific file data to recent commit

//checking different
git diff --> this command shows the difference between your last commit and current change

//branching
git checkout branch name --> this command is used to travel in between the branches
git checkout -b new_branch+name --> this command is used to create a new branch

//updating branch
git featch --> used to get or pull latest meta data
git pull origin branch_name -->gets the latest updated changes to the branch

//status and history
git status --> shows the current status of your working directory 

if shown by brown colour or (M) its is modified
if shown by Green colour or (U) its is modified
if shown by red colour or (D) its is modified

git log --> shows the commit history (detail log)
git log -p --> show the most recent commit with details
git log  --oneline --> show a compact history of the commit
